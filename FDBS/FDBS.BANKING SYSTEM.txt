

-- STEP 1: CREATE DATABASE

CREATE DATABASE IF NOT EXISTS Banking;
USE Banking;


-- =====================================================
-- STEP 2: CREATE TABLES
-- =====================================================

-- 1. CUSTOMER TABLE

CREATE TABLE Customer (
    CustomerID INT PRIMARY KEY AUTO_INCREMENT,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    Gender CHAR(1),
    DateOfBirth DATE,
    Phone VARCHAR(15) UNIQUE,
    Email VARCHAR(100) UNIQUE,
    Address VARCHAR(200)
);


-- 2. BRANCH TABLE

CREATE TABLE Branch (
    BranchID INT PRIMARY KEY AUTO_INCREMENT,
    BranchName VARCHAR(100) NOT NULL,
    Location VARCHAR(100) NOT NULL,
    Phone VARCHAR(15)
);


-- 3. EMPLOYEE TABLE

CREATE TABLE Employee (
    EmployeeID INT PRIMARY KEY AUTO_INCREMENT,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    Position VARCHAR(50),
    Phone VARCHAR(15),
    Email VARCHAR(100) UNIQUE,
    HireDate DATE,
    BranchID INT,

    FOREIGN KEY (BranchID)
    REFERENCES Branch(BranchID)
);


-- 4. ACCOUNT TABLE

CREATE TABLE Account (
    AccountNumber INT PRIMARY KEY AUTO_INCREMENT,
    AccountType VARCHAR(20) NOT NULL,
    Balance DECIMAL(12,2) DEFAULT 0.00,
    OpenDate DATE NOT NULL,
    AccountStatus VARCHAR(20) DEFAULT 'Active',
    CustomerID INT NOT NULL,
    BranchID INT NOT NULL,

    FOREIGN KEY (CustomerID)
    REFERENCES Customer(CustomerID),

    FOREIGN KEY (BranchID)
    REFERENCES Branch(BranchID),

    CHECK (Balance >= 0)
);


-- 5. BANK TRANSACTION TABLE

CREATE TABLE BankTransaction (
    TransactionID INT PRIMARY KEY AUTO_INCREMENT,
    TransactionType VARCHAR(20) NOT NULL,
    Amount DECIMAL(12,2) NOT NULL,
    TransactionDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    Description VARCHAR(200),
    AccountNumber INT NOT NULL,
    EmployeeID INT,

    FOREIGN KEY (AccountNumber)
    REFERENCES Account(AccountNumber),

    FOREIGN KEY (EmployeeID)
    REFERENCES Employee(EmployeeID),

    CHECK (Amount > 0)
);


-- 6. TRANSFER TABLE

CREATE TABLE Transfer (
    TransferID INT PRIMARY KEY AUTO_INCREMENT,
    FromAccount INT NOT NULL,
    ToAccount INT NOT NULL,
    Amount DECIMAL(12,2) NOT NULL,
    TransferDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    TransferStatus VARCHAR(20) DEFAULT 'Pending',

    FOREIGN KEY (FromAccount)
    REFERENCES Account(AccountNumber),

    FOREIGN KEY (ToAccount)
    REFERENCES Account(AccountNumber),

    CHECK (Amount > 0),
    CHECK (FromAccount <> ToAccount)
);


-- 7. LOAN TABLE

CREATE TABLE Loan (
    LoanID INT PRIMARY KEY AUTO_INCREMENT,
    LoanType VARCHAR(50) NOT NULL,
    LoanAmount DECIMAL(12,2) NOT NULL,
    InterestRate DECIMAL(5,2) NOT NULL,
    StartDate DATE,
    EndDate DATE,
    LoanStatus VARCHAR(20) DEFAULT 'Pending',
    CustomerID INT NOT NULL,

    FOREIGN KEY (CustomerID)
    REFERENCES Customer(CustomerID),

    CHECK (LoanAmount > 0),
    CHECK (InterestRate >= 0)
);


-- 8. PAYMENT TABLE

CREATE TABLE Payment (
    PaymentID INT PRIMARY KEY AUTO_INCREMENT,
    PaymentDate DATE NOT NULL,
    AmountPaid DECIMAL(12,2) NOT NULL,
    PaymentMethod VARCHAR(30),
    LoanID INT NOT NULL,

    FOREIGN KEY (LoanID)
    REFERENCES Loan(LoanID),

    CHECK (AmountPaid > 0)
);


-- =====================================================
-- STEP 3: CREATE INDEXES
-- =====================================================

CREATE INDEX idx_customer_name
ON Customer(FirstName, LastName);

CREATE INDEX idx_account_customer
ON Account(CustomerID);

CREATE INDEX idx_transaction_account
ON BankTransaction(AccountNumber);

CREATE INDEX idx_loan_customer
ON Loan(CustomerID);

CREATE INDEX idx_payment_loan
ON Payment(LoanID);

-- STEP 5: CREATE TRIGGERS
-- =====================================================

DELIMITER //


-- TRIGGER 1: ACCOUNT BALANCE VALIDATION

CREATE TRIGGER BeforeAccountInsert
BEFORE INSERT ON Account
FOR EACH ROW
BEGIN
    IF NEW.Balance < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Balance cannot be negative';
    END IF;
END //


-- TRIGGER 2: TRANSFER VALIDATION

CREATE TRIGGER BeforeTransferInsert
BEFORE INSERT ON Transfer
FOR EACH ROW
BEGIN
    IF NEW.FromAccount = NEW.ToAccount THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Source and destination cannot be same';
    END IF;

    IF NEW.Amount <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Transfer amount must be positive';
    END IF;
END //


-- TRIGGER 3: PAYMENT VALIDATION

CREATE TRIGGER BeforePaymentInsert
BEFORE INSERT ON Payment
FOR EACH ROW
BEGIN
    DECLARE v_LoanAmount DECIMAL(12,2);
    DECLARE v_TotalPaid DECIMAL(12,2);
    DECLARE v_Status VARCHAR(20);

    SELECT LoanAmount, LoanStatus
    INTO v_LoanAmount, v_Status
    FROM Loan
    WHERE LoanID = NEW.LoanID;

    SELECT COALESCE(SUM(AmountPaid), 0)
    INTO v_TotalPaid
    FROM Payment
    WHERE LoanID = NEW.LoanID;

    IF NEW.AmountPaid <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Payment must be positive';
    END IF;

    IF v_Status <> 'Approved' THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Loan is not approved';
    END IF;

    IF v_TotalPaid + NEW.AmountPaid > v_LoanAmount THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Payment exceeds loan amount';
    END IF;
END //


DELIMITER ;

-- PROCEDURE 1: DEPOSIT MONEY

CREATE PROCEDURE DepositMoney(
    IN p_AccountNumber INT,
    IN p_Amount DECIMAL(12,2)
)
BEGIN
    DECLARE v_Status VARCHAR(20);

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    IF p_Amount IS NULL OR p_Amount <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Invalid deposit amount';
    END IF;

    START TRANSACTION;

    SELECT AccountStatus
    INTO v_Status
    FROM Account
    WHERE AccountNumber = p_AccountNumber
    FOR UPDATE;

    IF v_Status <> 'Active' THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Account is not active';
    END IF;

    UPDATE Account
    SET Balance = Balance + p_Amount
    WHERE AccountNumber = p_AccountNumber;

    INSERT INTO BankTransaction
    (TransactionType, Amount, AccountNumber, Description)
    VALUES
    ('Deposit', p_Amount, p_AccountNumber, 'Cash deposit');

    COMMIT;
END //


-- PROCEDURE 2: WITHDRAW MONEY

CREATE PROCEDURE WithdrawMoney(
    IN p_AccountNumber INT,
    IN p_Amount DECIMAL(12,2)
)
BEGIN
    DECLARE v_Balance DECIMAL(12,2);
    DECLARE v_Status VARCHAR(20);

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    IF p_Amount IS NULL OR p_Amount <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Invalid withdrawal amount';
    END IF;

    START TRANSACTION;

    SELECT Balance, AccountStatus
    INTO v_Balance, v_Status
    FROM Account
    WHERE AccountNumber = p_AccountNumber
    FOR UPDATE;

    IF v_Status <> 'Active' THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Account is not active';
    END IF;

    IF v_Balance < p_Amount THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Insufficient balance';
    END IF;

    UPDATE Account
    SET Balance = Balance - p_Amount
    WHERE AccountNumber = p_AccountNumber;

    INSERT INTO BankTransaction
    (TransactionType, Amount, AccountNumber, Description)
    VALUES
    ('Withdrawal', p_Amount, p_AccountNumber,
     'Cash withdrawal');

    COMMIT;
END //


-- PROCEDURE 3: TRANSFER MONEY

CREATE PROCEDURE TransferMoney(
    IN p_FromAccount INT,
    IN p_ToAccount INT,
    IN p_Amount DECIMAL(12,2)
)
BEGIN
    DECLARE v_FromBalance DECIMAL(12,2);
    DECLARE v_FromStatus VARCHAR(20);
    DECLARE v_ToStatus VARCHAR(20);

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    IF p_FromAccount = p_ToAccount THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Cannot transfer to same account';
    END IF;

    IF p_Amount IS NULL OR p_Amount <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Invalid transfer amount';
    END IF;

    START TRANSACTION;

    SELECT Balance, AccountStatus
    INTO v_FromBalance, v_FromStatus
    FROM Account
    WHERE AccountNumber = p_FromAccount
    FOR UPDATE;

    SELECT AccountStatus
    INTO v_ToStatus
    FROM Account
    WHERE AccountNumber = p_ToAccount
    FOR UPDATE;

    IF v_FromStatus <> 'Active' OR v_ToStatus <> 'Active' THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Both accounts must be active';
    END IF;

    IF v_FromBalance < p_Amount THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Insufficient balance';
    END IF;

    UPDATE Account
    SET Balance = Balance - p_Amount
    WHERE AccountNumber = p_FromAccount;

    UPDATE Account
    SET Balance = Balance + p_Amount
    WHERE AccountNumber = p_ToAccount;

    INSERT INTO Transfer
    (FromAccount, ToAccount, Amount, TransferStatus)
    VALUES
    (p_FromAccount, p_ToAccount, p_Amount, 'Completed');

    INSERT INTO BankTransaction
    (TransactionType, Amount, AccountNumber, Description)
    VALUES
    ('Transfer Out', p_Amount, p_FromAccount, 'Money transferred out'),
    ('Transfer In', p_Amount, p_ToAccount, 'Money received');

    COMMIT;
END //


-- PROCEDURE 4: MAKE LOAN PAYMENT

CREATE PROCEDURE MakeLoanPayment(
    IN p_LoanID INT,
    IN p_Amount DECIMAL(12,2),
    IN p_Method VARCHAR(30)
)
BEGIN
    DECLARE v_LoanAmount DECIMAL(12,2);
    DECLARE v_TotalPaid DECIMAL(12,2);
    DECLARE v_Status VARCHAR(20);

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    IF p_Amount IS NULL OR p_Amount <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Invalid payment amount';
    END IF;

    START TRANSACTION;

    SELECT LoanAmount, LoanStatus
    INTO v_LoanAmount, v_Status
    FROM Loan
    WHERE LoanID = p_LoanID
    FOR UPDATE;

    SELECT COALESCE(SUM(AmountPaid), 0)
    INTO v_TotalPaid
    FROM Payment
    WHERE LoanID = p_LoanID;

    IF v_Status <> 'Approved' THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Loan is not approved';
    END IF;

    IF v_TotalPaid + p_Amount > v_LoanAmount THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Payment exceeds remaining principal';
    END IF;

    INSERT INTO Payment
    (PaymentDate, AmountPaid, PaymentMethod, LoanID)
    VALUES
    (CURDATE(), p_Amount, p_Method, p_LoanID);

    UPDATE Loan
    SET LoanStatus =
        CASE
            WHEN v_TotalPaid + p_Amount = v_LoanAmount
            THEN 'Paid'
            ELSE 'Approved'
        END
    WHERE LoanID = p_LoanID;

    COMMIT;
END //

-- VIEW 1: CUSTOMER ACCOUNT VIEW

CREATE VIEW CustomerAccountView AS
SELECT
    C.CustomerID,
    C.FirstName,
    C.LastName,
    C.Phone,
    A.AccountNumber,
    A.AccountType,
    A.Balance,
    A.AccountStatus
FROM Customer C
JOIN Account A
ON C.CustomerID = A.CustomerID;


-- VIEW 2: LOAN REPORT VIEW

CREATE VIEW LoanReportView AS
SELECT
    L.LoanID,
    C.FirstName,
    C.LastName,
    L.LoanType,
    L.LoanAmount,
    L.InterestRate,
    L.LoanStatus
FROM Loan L
JOIN Customer C
ON L.CustomerID = C.CustomerID;


-- VIEW 3: TRANSACTION REPORT VIEW

CREATE VIEW TransactionReportView AS
SELECT
    T.TransactionID,
    T.TransactionType,
    T.Amount,
    T.TransactionDate,
    A.AccountNumber,
    C.FirstName,
    C.LastName
FROM BankTransaction T
JOIN Account A
ON T.AccountNumber = A.AccountNumber
JOIN Customer C
ON A.CustomerID = C.CustomerID;

-- 20. REMAINING LOAN PRINCIPAL

SELECT
    L.LoanID,
    L.LoanAmount,
    COALESCE(SUM(P.AmountPaid), 0) AS TotalPaid,
    L.LoanAmount - COALESCE(SUM(P.AmountPaid), 0)
        AS RemainingPrincipal
FROM Loan L
LEFT JOIN Payment P
ON L.LoanID = P.LoanID
GROUP BY L.LoanID, L.LoanAmount;
